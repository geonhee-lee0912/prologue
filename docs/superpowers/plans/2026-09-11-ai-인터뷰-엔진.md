```ts
/**
 * 민감정보 탐지·마스킹 (개인정보보호법 제23조)
 *
 * AI 인터뷰는 자유 대화라서 우리가 묻지 않아도 민감정보가 들어온다.
 * 저장 전에 반드시 이 함수를 거친다.
 *
 * 주의: 마스킹은 우리 DB 만 보호한다. 실시간 대화라 LLM API 전송 시점은
 * 막을 수 없으므로, 국외이전 동의 범위에 민감정보를 포함시켜야 한다.
 *
 * MVP 는 규칙(키워드) 기반이다. 재현율이 낮으므로 추후 LLM 기반 2차
 * 탐지를 덧붙인다.
 */

export enum SensitiveCategory {
  RELIGION = 'RELIGION',
  HEALTH = 'HEALTH',
  SEXUAL = 'SEXUAL',
}

const KEYWORDS: Record<SensitiveCategory, string[]> = {
  [SensitiveCategory.RELIGION]: [
    '교회', '성당', '절에', '불교', '기독교', '천주교', '개신교', '이슬람',
    '목사', '신부님', '스님', '예배', '미사', '신앙', '모태신앙', '무교',
    '전도', '선교', '교인', '성경', '불경',
  ],
  [SensitiveCategory.HEALTH]: [
    '우울증', '공황장애', '조울증', '불안장애', '조현병', 'ADHD',
    '암 진단', '항암', '수술', '입원', '지병', '만성질환', '장애등급',
    '정신과', '상담치료', '약을 먹', '복용 중', '치료를 받',
    '당뇨', '고혈압', '갑상선',
  ],
  [SensitiveCategory.SEXUAL]: [
    '성관계', '잠자리', '성경험', '성적 취향', '피임', '임신중절',
    '낙태', '성병', '동성애', '양성애', '성정체성',
  ],
};

const PLACEHOLDER: Record<SensitiveCategory, string> = {
  [SensitiveCategory.RELIGION]: '[민감정보:종교]',
  [SensitiveCategory.HEALTH]: '[민감정보:건강]',
  [SensitiveCategory.SEXUAL]: '[민감정보:성생활]',
};

export interface MaskSensitiveOptions {
  /**
   * 민감정보 별도 동의를 받았고, 종교를 매칭에 쓰기로 한 경우 true.
   * 종교만 보존되며 건강·성생활은 동의와 무관하게 항상 마스킹한다.
   */
  allowReligion: boolean;
}

export interface MaskSensitiveResult {
  masked: string;
  categories: SensitiveCategory[];
}

export function maskSensitive(
  text: string,
  options: MaskSensitiveOptions,
): MaskSensitiveResult {
  let masked = text;
  const categories: SensitiveCategory[] = [];

  for (const category of Object.values(SensitiveCategory)) {
    if (category === SensitiveCategory.RELIGION && options.allowReligion) {
      continue;
    }

    let hit = false;
    for (const keyword of KEYWORDS[category]) {
      if (masked.includes(keyword)) {
        hit = true;
        masked = masked.split(keyword).join(PLACEHOLDER[category]);
      }
    }
    if (hit) {
      categories.push(category);
    }
  }

  return { masked, categories };
}
```

```ts
/**
 * LLM API 전송 전 식별자 제거 (CLAUDE.md §10.2c)
 *
 * AI 는 실명을 알 필요가 없다. 국외이전 부담을 줄이고
 * "깊은 것은 AI 만 본다" 원칙과도 같은 방향이다.
 */

export interface PseudonymizeOptions {
  /** 사용자의 실명. 알고 있으면 대화 중 등장해도 치환한다. */
  name?: string;
}

const PHONE_WITH_HYPHEN = /01[016789]-\d{3,4}-\d{4}/g;
const PHONE_PLAIN = /01[016789]\d{7,8}/g;
const EMAIL = /[\w.+-]+@[\w-]+\.[\w.-]+/g;
const RRN = /\d{6}-[1-4]\d{6}/g;

export function pseudonymize(
  text: string,
  options: PseudonymizeOptions,
): string {
  let result = text;

  // 주민번호를 먼저 — 휴대폰 패턴과 겹치지 않도록
  result = result.replace(RRN, '[주민번호]');
  result = result.replace(EMAIL, '[이메일]');
  result = result.replace(PHONE_WITH_HYPHEN, '[연락처]');
  result = result.replace(PHONE_PLAIN, '[연락처]');

  if (options.name && options.name.length >= 2) {
    result = result.split(options.name).join('[이름]');
  }

  return result;
}
```

```ts
import type {
  LlmCompletionRequest,
  LlmCompletionResult,
  LlmProvider,
} from '@prologue/shared';

/**
 * OpenAI GPT-5.6 Luna 구현체.
 *
 * 주의 (CLAUDE.md §10.2c, §13):
 * - 여기에 도달하는 텍스트는 이미 가명처리되어 있어야 한다
 * - 국외이전이므로 별도 동의를 받은 사용자만 호출 대상이다
 */
export class OpenAiLlmService implements LlmProvider {
  private readonly endpoint = 'https://api.openai.com/v1/chat/completions';

  constructor(
    private readonly apiKey: string,
    private readonly model: string,
  ) {}

  async complete(
    request: LlmCompletionRequest,
  ): Promise<LlmCompletionResult> {
    const response = await fetch(this.endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${this.apiKey}`,
      },
      body: JSON.stringify({
        model: this.model,
        messages: request.messages,
        max_completion_tokens: request.maxOutputTokens ?? 800,
      }),
    });

    if (!response.ok) {
      const detail = await response.text();
      throw new Error(`LLM 호출 실패 (${response.status}): ${detail}`);
    }

    const body = await response.json();
    return {
      content: body.choices?.[0]?.message?.content ?? '',
      inputTokens: body.usage?.prompt_tokens ?? 0,
      outputTokens: body.usage?.completion_tokens ?? 0,
    };
  }
}
```

```ts
        state.sessions.set(where.id, next);

        const row = { id: nextId(), ...data };
```

```ts
      );

      data: { participantId },

        sessionId: session.id,

        maskedCategories: [],
        sequence: 0,
      },
    });
```

```ts
  async submitAnswer(
    sessionId: string,
    answer: string,
  ): Promise<SubmitResult> {
    const session = await this.prisma.interviewSession.findUnique({
      where: { id: sessionId },
    });
    if (!session) {
      throw new NotFoundException('세션을 찾을 수 없습니다');
    }
    if (session.status !== 'in_progress') {
      throw new BadRequestException('이미 종료된 세션입니다');
    }

    const participant = await this.prisma.participant.findUnique({
      where: { id: session.participantId },
    });

    const masked = maskSensitive(answer, {
      allowReligion: Boolean(participant?.consentSensitiveAt),
    });

    const existingTurns = await this.prisma.interviewTurn.findMany({
      where: { sessionId },
    });

    await this.prisma.interviewTurn.create({
      data: {
        sessionId,
        role: 'user',
        content: masked.masked,
        maskedCategories: masked.categories.map((c) => c.toLowerCase()),
        sequence: existingTurns.length,
      },
    });

    const turnCount = session.turnCount + 1;
    const completed = turnCount >= MAX_TURNS;

    const history: LlmMessage[] = existingTurns.map((t) => ({
      role: t.role === 'user' ? 'user' : 'assistant',
      content: pseudonymize(t.content, {}),
    }));
    history.push({ role: 'user', content: pseudonymize(answer, {}) });

    const completion = await this.llm.complete({
      messages: [
        {
          role: 'system',
          content: buildInterviewSystemPrompt({ turnCount }),
        },
        ...history,
      ],
    });

    await this.prisma.interviewTurn.create({
      data: {
        sessionId,
        role: 'assistant',
        content: completion.content,
        maskedCategories: [],
        sequence: existingTurns.length + 1,
      },
    });

    await this.prisma.interviewSession.update({
      where: { id: sessionId },
      data: {
        turnCount,
        ...(completed
          ? { status: 'completed', completedAt: new Date() }
          : {}),
      },
    });

    if (completed) {
      await this.generateReport(sessionId);
    }
```
